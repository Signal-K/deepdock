```dataviewjs
// Get current date from filename
const currentDate = dv.current().file.name;

if (currentDate && currentDate.match(/^\d{4}-\d{2}-\d{2}$/)) {
  // Get all daily files and extract dates from filenames
  const dailyFiles = dv.pages('"Operations/Daily/Active"')
    .filter(p => p.file.name.match(/^\d{4}-\d{2}-\d{2}$/))
    .map(p => ({ date: p.file.name, path: p.file.path }))
    .sort((a, b) => a.date.localeCompare(b.date));
  
  const currentIndex = dailyFiles.findIndex(f => f.date === currentDate);
  
  let prevButton = "⬅️ Previous";
  let nextButton = "Next ➡️";
  
  if (currentIndex > 0) {
    const prevFile = dailyFiles[currentIndex - 1];
    prevButton = `[⬅️ ${prevFile.date}](${prevFile.path.replace('content/', '')})`;
  }
  
  if (currentIndex < dailyFiles.length - 1) {
    const nextFile = dailyFiles[currentIndex + 1];
    nextButton = `[${nextFile.date} ➡️](${nextFile.path.replace('content/', '')})`;
  }
  
  dv.paragraph(`${prevButton} | ${nextButton}`);
} else {
  dv.paragraph("⬅️ Previous | Next ➡️");
}
```

## 🗓️ Today's Focus Tasks

```dataviewjs
const today = dv.current().date ? dv.current().date.toString().split('T')[0] : null;
if (!today) {
  dv.paragraph('No date set for this page.');
} else {
  let tasks = [];
  for (let page of dv.pages('"content/Projects"')) {
    if (page.file && page.file.tasks) {
      for (let t of page.file.tasks) {
        if (t.text.includes(`#${today}`)) {
          tasks.push(t);
        }
      }
    }
  }
  if (tasks.length === 0) {
    dv.paragraph('No tasks tagged for today.');
  } else {
    dv.taskList(tasks, false);
  }
}
```

## 🧩 In-Progress Story Tasks

```dataviewjs
// Stories in progress with no tasks
// Find story pages under content/Categories and filter those marked in-progress
const pages = dv.pages("\"content/Categories\"").filter(p => p.file && p.file.path && p.file.path.includes('/Stories/'));
const allStories = pages.filter(p => p.status && String(p.status).toLowerCase() === 'in-progress');

// Collect pages that actually have tasks
const withTasks = new Set(pages.filter(p => p.file && p.file.tasks).map(p => p.file.path));
const noTaskStories = allStories.filter(p => !withTasks.has(p.file.path));

dv.header(3, "📝 In-Progress Stories (No Tasks)");
if (!noTaskStories || noTaskStories.length === 0) {
  dv.paragraph("No in-progress stories without tasks.");
} else {
  dv.header(4, "Stories in progress (no tasks defined):");
  noTaskStories.forEach(p => dv.paragraph(p.file.link));
}
```
